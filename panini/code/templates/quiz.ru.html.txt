<script>
"use strict";
const BANK = JSON.parse(document.getElementById('bank').textContent);
const META = __META__;
const $ = id => document.getElementById(id);
const esc = s => String(s).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));

$('bankinfo').textContent = 'Банк: ' + META.core_terms + ' терминов, '
  + META.russian_glosses + ' русских толкований; форма A — '
  + META.forms.A.total + ' заданий, форма B — ' + META.forms.B.total + ' заданий.';

$('consent').addEventListener('change', () => { $('start').disabled = !$('consent').checked; });

// Deterministic per-participant shuffle so pre/post orders are reproducible.
function seeded(str){ let h=2166136261; for(const c of str){h^=c.charCodeAt(0);h=Math.imul(h,16777619);} 
  return function(){h+=0x6D2B79F5;let t=h;t=Math.imul(t^t>>>15,t|1);t^=t+Math.imul(t^t>>>7,t|61);
    return ((t^t>>>14)>>>0)/4294967296;}; }
function shuffled(a,rnd){a=a.slice();for(let i=a.length-1;i>0;i--){const j=Math.floor(rnd()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;}

let CTX=null, ITEMS=[], CUR=0, RESP=[], T0=0;

$('start').addEventListener('click', () => {
  const pid = $('pid').value.trim().padStart(4,'0') || 'ANON';
  const arm = $('arm').value, form = $('form').value, wave = $('wave').value;
  CTX = {pid, arm, form, wave, started:new Date().toISOString()};
  const rnd = seeded(pid+'|'+form+'|'+wave);
  ITEMS = shuffled(BANK.filter(i=>i.form===form), rnd);
  CUR=0; RESP=[];
  $('setup').hidden=true; $('done').hidden=true; $('test').hidden=false;
  render();
});

function render(){
  if(CUR>=ITEMS.length){ return finish(); }
  const it=ITEMS[CUR];
  $('pos').textContent='Задание '+(CUR+1)+' из '+ITEMS.length;
  $('clock').textContent = ({gloss:'значение термина',implicit_trained:'неписаный термин · обученный материал',
    implicit_transfer:'неписаный термин · перенос'})[it.type] || it.type;
  $('fill').style.width=(100*CUR/ITEMS.length)+'%';
  let h='<div><span class="tag">'+esc(it.type)+'</span><span class="tag">блок B'+it.batch+'</span>';
  if(it.channel) h+='<span class="tag">'+esc(it.channel)+'</span>';
  h+='</div><p>'+it.prompt+'</p>';
  if(it.sutra_ref){
    h+='<div class="sutra"><div class="ref">Aṣṭādhyāyī '+esc(it.sutra_ref)+'</div>'
      +'<div class="deva">&#9673; '+esc(it.sutra_deva)+'</div>'
      +'<div class="pc">padaccheda: [ &#9633; ] '+esc(it.padaccheda)+'</div></div>';
  } else {
    h+='<div class="sutra deva">'+esc(it.term_deva)+' &nbsp;<span style="font-size:15px;color:var(--muted)">'+esc(it.term_iast)+'</span></div>';
  }
  h+='</div>';
  $('qbody').innerHTML=h;
  const box=$('opts'); box.innerHTML='';
  it.options.forEach((o,k)=>{
    const b=document.createElement('button'); b.className='opt'; b.textContent=o;
    b.addEventListener('click',()=>answer(k)); box.appendChild(b);
  });
  T0=performance.now();
}

function answer(k){
  const it=ITEMS[CUR];
  RESP.push({participant:CTX.pid,arm:CTX.arm,form:CTX.form,wave:CTX.wave,
    item_id:it.id,type:it.type,batch:it.batch,term_iast:it.term_iast,
    sutra_ref:it.sutra_ref,channel:it.channel,answer_index:k,correct_index:it.answer,
    is_correct:k===it.answer?1:0,rt_ms:Math.round(performance.now()-T0),
    timestamp:new Date().toISOString()});
  CUR++; render();
}

function finish(){
  $('test').hidden=true; $('done').hidden=false;
  const groups={}; for(const r of RESP){ (groups[r.type]=groups[r.type]||[]).push(r); }
  let h='<p><b>Всего верно: '+RESP.filter(r=>r.is_correct).length+' из '+RESP.length+'</b></p><table><tr><th>Тип задания</th><th>Верно</th><th>Из</th><th>Среднее время, мс</th></tr>';
  for(const k of Object.keys(groups)){
    const g=groups[k], ok=g.filter(r=>r.is_correct).length,
      rt=Math.round(g.reduce((s,r)=>s+r.rt_ms,0)/g.length);
    h+='<tr><td>'+esc(k)+'</td><td>'+ok+'</td><td>'+g.length+'</td><td>'+rt+'</td></tr>';
  }
  h+='</table><p class="note">Первичная конечная точка протокола — прирост по подшкале '
    +'<code>implicit_trained + implicit_transfer</code> относительно pretest; '
    +'задания типа <code>gloss</code> служат ковариатом базового словарного запаса.</p>';
  $('result').innerHTML=h;
}

$('csv').addEventListener('click',()=>{
  const cols=['participant','arm','form','wave','item_id','type','batch','term_iast',
    'sutra_ref','channel','answer_index','correct_index','is_correct','rt_ms','timestamp'];
  const q=v=>'"'+String(v==null?'':v).replace(/"/g,'""')+'"';
  const lines=[cols.join(',')].concat(RESP.map(r=>cols.map(c=>q(r[c])).join(',')));
  const blob=new Blob(['\ufeff'+lines.join('\r\n')],{type:'text/csv;charset=utf-8'});
  const a=document.createElement('a');
  a.href=URL.createObjectURL(blob);
  a.download='panini_'+CTX.pid+'_'+CTX.form+'_'+CTX.wave+'.csv';
  document.body.appendChild(a); a.click(); document.body.removeChild(a);
  setTimeout(()=>URL.revokeObjectURL(a.href),2000);
});
$('again').addEventListener('click',()=>{ $('done').hidden=true; $('setup').hidden=false; });
</script>
